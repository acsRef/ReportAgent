### 2. Tracer 双 sink

升级 `Tracer.flush()` 从 `_flush_db` → `_flush`（双 sink）：

```python
async def _flush(self):
    """双 sink：PG repository + Langfuse SDK。两者失败都 best-effort。"""
    # Sink 1: PG (always)
    await self._flush_pg()
    # Sink 2: Langfuse (env-gated)
    if langfuse_config.enabled:
        await self._flush_langfuse()

async def _flush_pg(self):
    """原有 _flush_db 逻辑，保持不变。"""
    # ... save_trace / save_span / save_llm_call

async def _flush_langfuse(self):
    """把 in-memory spans/llm_calls 转成 Langfuse event。"""
    langfuse = get_langfuse_client()
    if not langfuse:
        return
    try:
        # 用 trace_id 作为 Langfuse trace_id（关联 PG / Langfuse）
        with langfuse.start_as_current_span(
            name="report_agent_run",
            id=self.trace_id,
        ) as root:
            for span in self._spans:
                with langfuse.start_as_current_span(
                    name=span.span_name,
                    input=_redact(span.input) if span.input else None,
                ):
                    # Langfuse 自动记录 latency
                    pass
            for llm_call in self._llm_calls:
                langfuse.update_current_observation(
                    model=llm_call.model,
                    usage={
                        "input": llm_call.prompt_tokens,
                        "output": llm_call.completion_tokens,
                    },
                )
            for pv in self._prompt_versions:
                langfuse.update_current_observation(
                    metadata={"prompt_name": pv["name"], "prompt_version": pv["version"]},
                )
        await asyncio.wait_for(
            asyncio.to_thread(langfuse.flush),
            timeout=langfuse_config.flush_timeout,
        )
    except (asyncio.TimeoutError, Exception) as exc:
        logger.warning("langfuse flush failed for trace_id=%s: %s", self.trace_id, exc)
```

**设计要点**：
- 双 sink 各自独立失败兜底（best-effort）
- Langfuse SDK 调用走 `asyncio.to_thread`（SDK 是 sync HTTP）
- 用 `asyncio.wait_for` 兜底 flush timeout
- trace_id 一致（PG + Langfuse 双向可关联）

### 3. PII redaction 层（adapter 前一层）

新建 `backend/app/observability/redaction.py`：

```python
"""PII redaction：所有进 tracer / Langfuse / log 的 user_query / span input / LLM call input
统一过 mask_pii。
"""
from app.utils.pii import mask_pii


def redact(value: Any) -> Any:
    """递归 mask dict / list / str 中的 PII。"""
    if isinstance(value, str):
        return mask_pii(value)
    if isinstance(value, dict):
        return {k: redact(v) for k, v in value.items()}
    if isinstance(value, list):
        return [ redact(v) for v in value ]
    return value


def redact_user_query(query: str) -> str:
    """user_query 入口 redaction。"""
    return mask_pii(query or "")
```

**集成点**：
- `Tracer._summarize_state(state)` 进 span input 前
- `_handle_output_span(tracer, result)` 进 span input 前
- `add_llm_call(...)` 的 prompt_tokens 不动（数字本身无 PII），但 LLM 实际 prompt 文本进 Langfuse metadata 时也走 redact
- LangChain CallbackHandler 收到 LLM input 时自动 redact

**复用现有**：`backend/app/utils/pii.py` `mask_pii` 已实现正则族（CLAUDE.md §468 明确指定 P13 复用）。

### 5. LangChain CallbackHandler 自动接入（可选增强）

Langfuse 提供 `CallbackHandler`，可直接传给 LangChain 的 `chain.invoke(..., config={"callbacks": [handler]})`，自动捕获每次 LLM 调用。

但 ReportAgent 当前 `LLMAdapter` 用裸 `ChatOpenAI` 不走 LangChain callbacks（adapter.py:75-105 直接调 `llm.invoke(prompt)`，未传 config.callbacks）—— **本 plan 不引入 LangChain callback 改造**（adapter 是宪法接口 P6 锁定）。

**取而代之**：tracer.flush 时把已记录到 `self._llm_calls` 的数据（model / tokens / latency）显式写 Langfuse metadata（不靠 LangChain 自动捕获）。

### 6. Prompt version + Model 进 Langfuse

`add_prompt_version(name, version)` 已存本地（P7 D3 落地），本 plan 在 `_flush_langfuse` 时把 `_prompt_versions` 写到 Langfuse observation metadata（key=`prompt_name` / `prompt_version`）。

`add_llm_call` 已存本地 model 字段，`_flush_langfuse` 写到 `model=` 参数。

```python
from __future__ import annotations

import os

from pydantic import BaseModel


class LangfuseConfig(BaseModel):
    """Langfuse SDK 配置（env-gated）。

    关键：enabled=False 时整个 Langfuse 路径跳过，仅 PG sink 生效。
    """
    enabled: bool = False
    public_key: str | None = None
    secret_key: str | None = None
    host: str = "https://cloud.langfuse.com"
    flush_timeout: float = 10.0

    def __init__(self, **data):
        super().__init__(**data)
        if not self.enabled:
            # env 探测
            pk = os.getenv("LANGFUSE_PUBLIC_KEY")
            sk = os.getenv("LANGFUSE_SECRET_KEY")
            if pk and sk:
                self.enabled = True
                self.public_key = pk
                self.secret_key = sk
        self.host = os.getenv("LANGFUSE_HOST", "https://cloud.langfuse.com")
        self.flush_timeout = float(os.getenv("LANGFUSE_FLUSH_TIMEOUT", "10.0"))
```

新建 `backend/app/observability/langfuse_client.py`：
```python
"""Langfuse SDK 单例（fail-open）。"""
from __future__ import annotations

import logging
from typing import Any

from app.observability.langfuse_config import LangfuseConfig

logger = logging.getLogger(__name__)

_client: Any | None = None


def get_langfuse_client() -> Any | None:
    """返回 Langfuse SDK client 单例；enabled=False 或 init 失败 → None。"""
    global _client
    cfg = LangfuseConfig()
    if not cfg.enabled:
        return None
    if _client is None:
        try:
            from langfuse import Langfuse  # 延迟 import，未装 langfuse 时优雅降级
            _client = Langfuse(
                public_key=cfg.public_key,
                secret_key=cfg.secret_key,
                host=cfg.host,
            )
        except Exception as exc:
            logger.warning("langfuse init failed: %s", exc)
            _client = None
    return _client
```

```python
from __future__ import annotations

from typing import Any

from app.utils.pii import mask_pii


def redact(value: Any) -> Any:
    """递归 mask dict / list / str 中的 PII。"""
    if isinstance(value, str):
        return mask_pii(value)
    if isinstance(value, dict):
        return {k: redact(v) for k, v in value.items()}
    if isinstance(value, list):
        return [redact(v) for v in value]
    return value

    """user_query 入口 redaction（空串 / None 安全）。"""
    return mask_pii(query or "")
```

```python
"""tracer → Langfuse 转换器。

把 in-memory spans / llm_calls / prompt_versions 转成 Langfuse SDK 调用。
PII 在入参前过 redact()；Langfuse SDK 失败 best-effort 不抛。
"""
from __future__ import annotations

import asyncio
import logging
from typing import TYPE_CHECKING, Any

from app.observability.langfuse_client import get_langfuse_client
from app.observability.langfuse_config import LangfuseConfig
from app.observability.redaction import redact

if TYPE_CHECKING:
    from app.infra.trace.sdk import Tracer

logger = logging.getLogger(__name__)


async def flush_to_langfuse(
    tracer: "Tracer",
    langfuse_config: LangfuseConfig | None = None,
) -> None:
    """把 tracer 的 spans/llm_calls/prompt_versions 转 Langfuse SDK 调用。

    Best-effort：任何异常都 logger.warning 不抛。
    """
    cfg = langfuse_config or LangfuseConfig()
    if not cfg.enabled:
        return

    langfuse = get_langfuse_client()
    if langfuse is None:
        return

    try:
        # root span
        with langfuse.start_as_current_span(
            name="report_agent_run",
            id=tracer.trace_id,
            input=redact({"user_query": tracer.user_query}) if tracer.user_query else None,
        ) as root:
            # 子 span
            for span in tracer._spans:
                with langfuse.start_as_current_span(
                    name=span.span_name,
                    input=redact(span.input) if getattr(span, "input", None) else None,
                ):
                    # Langfuse 自动记录 latency
                    pass
            # LLM calls
            for llm_call in tracer._llm_calls:
                try:
                    langfuse.update_current_observation(
                        model=llm_call.model,
                        usage={
                            "input": llm_call.prompt_tokens,
                            "output": llm_call.completion_tokens,
                        },
                    )
                except Exception as exc:
                    logger.warning("langfuse update llm_call failed: %s", exc)
            # Prompt versions
            for pv in tracer._prompt_versions:
                try:
                    langfuse.update_current_observation(
                        metadata={
                            "prompt_name": pv.get("name"),
                            "prompt_version": pv.get("version"),
                        },
                    )
                except Exception as exc:
                    logger.warning("langfuse update prompt_version failed: %s", exc)
        # flush with timeout
        try:
            await asyncio.wait_for(
                asyncio.to_thread(langfuse.flush),
                timeout=cfg.flush_timeout,
            )
        except asyncio.TimeoutError:
            logger.warning(
                "langfuse flush timed out after %ss for trace_id=%s",
                cfg.flush_timeout, tracer.trace_id,
            )
    except Exception as exc:
        logger.warning(
            "langfuse flush failed for trace_id=%s: %s", tracer.trace_id, exc,
        )
```
